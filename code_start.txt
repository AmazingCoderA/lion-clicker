        }
        else if (cmd == "0")

void ChangeBind()
{
    Console.Clear();
    Console.WriteLine(T("bind_title"));
    Console.Write($"{T("bind_taps")} ");
    string taps = Console.ReadLine() ?? "";
    int count = 1;
    if (int.TryParse(taps, out int tc))
    {
        count = Math.Clamp(tc, 1, 5);
    }

    Console.Write($"{T("bind_timeout")} ");
    string timeoutStr = Console.ReadLine() ?? "";
    if (!string.IsNullOrEmpty(timeoutStr) && int.TryParse(timeoutStr, out int to) && to >= 10)
    {
        config.triggerTimeoutMs = to;
    }

    Console.WriteLine(T("bind_key"));
    int key = ReadVirtualKey();
    if (key == -1) return;

    config.triggerKey = key;
    config.triggerCount = count;
    SaveConfig();

    Console.WriteLine($"{T("bind_saved")} ({GetKeyName(key)} x{count}, timeout {config.triggerTimeoutMs}ms)");
    Thread.Sleep(1500);
}

void RunClickerEngine()
{
    Console.Clear();
    Console.WriteLine($"=== {T("engine_running")} ===");
    Console.WriteLine($"{T("engine_hotkey")} {GetKeyName(config.triggerKey)} x{config.triggerCount} (timeout {config.triggerTimeoutMs}ms)");
    Console.WriteLine("---------------------------------------");
    Console.WriteLine(T("engine_toggle"));
    Console.WriteLine(T("engine_esc"));
    Console.WriteLine("---------------------------------------");

    bool active = false;
    int tapCount = 0;
    DateTime lastTapTime = DateTime.UtcNow;
    bool prevKeyState = false;

    // Clear input buffer
    while (Console.KeyAvailable) Console.ReadKey(true);

    bool CheckEscape()
    {
        while (Console.KeyAvailable)
        {
            if (Console.ReadKey(true).Key == ConsoleKey.Escape) return true;
        }
        return false;
    }

    bool WaitAndPoll(int ms)
    {
        int slept = 0;
        while (slept < ms)
        {
            if (CheckEscape()) return true;

            bool cState = (GetAsyncKeyState(config.triggerKey) & 0x8000) != 0;
            if (cState && !prevKeyState)
            {
                DateTime now = DateTime.UtcNow;
                double msPassed = (now - lastTapTime).TotalMilliseconds;
                if (msPassed < config.triggerTimeoutMs) tapCount++;
                else tapCount = 1;
                lastTapTime = now;

                if (tapCount >= config.triggerCount)
                {
                    active = !active;
                    Console.WriteLine(active ? T("engine_state_on") : T("engine_state_off"));
                    tapCount = 0;
                    prevKeyState = cState;
                    return false; // Interrupted sleep due to state toggle
                }
            }
            prevKeyState = cState;

            if ((DateTime.UtcNow - lastTapTime).TotalMilliseconds > config.triggerTimeoutMs)
            {
                tapCount = 0;
            }

            int toSleep = Math.Min(10, ms - slept);
            Thread.Sleep(toSleep);
            slept += toSleep;

            // If active turned off from another place, break sleep
            if (!active && ms > 10) return false;
        }
        return false;
    }

    bool PerformSequence()
    {
        int holdBase = config.antiDetect ? config.holdTime : 10;
        int holdRange = config.antiDetect ? config.holdRange : 0;
        int clicks = 1;

        switch (config.clickPattern)
        {
            case ClickPattern.Single: clicks = 1; break;
            case ClickPattern.Double: clicks = 2; break;
            case ClickPattern.Triple: clicks = 3; break;
            case ClickPattern.Burst: clicks = config.burstCount; break;
        }

        for (int i = 0; i < clicks; ++i)
        {
            if (!active) break;

            if (config.simulateMove) CursorMoveRandom(config.moveRadius);
            int hold = holdBase;
            if (holdRange > 0) hold += rng.Next(-holdRange, holdRange + 1);
            if (hold < 1) hold = 1;

            MouseClick(config.clickButton, hold);

            if (i < clicks - 1)
            {
                int interval = config.burstInterval;
                if (config.antiDetect && config.burstIntervalRange > 0)
                {
                    interval += rng.Next(-config.burstIntervalRange, config.burstIntervalRange + 1);
                    if (interval < 1) interval = 1;
                }
                if (WaitAndPoll(interval)) return true;
            }
        }
        if (config.simulateMove && active) CursorMoveRandom(config.moveRadius);
        return false;
    }

    while (true)
    {
        if (WaitAndPoll(10)) return;

        if (active)
        {
            if (PerformSequence()) return;

            if (config.antiDetect && config.randomBreakEnabled)
            {
                if (rng.Next(0, 100) < config.randomBreakProbability)
                {
                    int breakMs = rng.Next(config.randomBreakMin, config.randomBreakMax);
                    if (WaitAndPoll(breakMs)) return;
                    if (!active) continue;
                }
            }

            int delay = config.baseDelay;
            if (config.antiDetect)
            {
                if (config.randomDist == RandomDist.Uniform)
                {
                    delay += rng.Next(-config.randomRange, config.randomRange + 1);
                }
                else
                {
                    double g = GetNormalRandom(0, config.randomRange / 3.0);
                    delay += (int)Math.Round(g);
                }
                if (delay < 1) delay = 1;
            }

            if (WaitAndPoll(delay)) return;
        }
    }
}

void MouseClick(ClickType type, int holdMs)
{
    uint down, up;
    switch (type)
    {
        case ClickType.Right: down = 0x08; up = 0x10; break;   // R_DOWN, R_UP
        case ClickType.Middle: down = 0x20; up = 0x40; break;  // M_DOWN, M_UP
        default: down = 0x02; up = 0x04; break;                // L_DOWN, L_UP
    }

    mouse_event(down, 0, 0, 0, UIntPtr.Zero);
    if (holdMs > 0) Thread.Sleep(holdMs);
    mouse_event(up, 0, 0, 0, UIntPtr.Zero);
}

void CursorMoveRandom(int radius)
{
    if (radius <= 0) return;
    int dx = rng.Next(-radius, radius + 1);
    int dy = rng.Next(-radius, radius + 1);
    mouse_event(0x0001, dx, dy, 0, UIntPtr.Zero); // MOUSEEVENTF_MOVE = 0x0001
}

int ReadVirtualKey()
{
    Console.WriteLine(T("press_key"));
    bool anyDown = true;
    while (anyDown)
    {
        anyDown = false;
        foreach (int vk in ALL_KEYS)
        {
            if ((GetAsyncKeyState(vk) & 0x8000) != 0) { anyDown = true; break; }
        }
        if (anyDown) Thread.Sleep(10);
    }

    while (Console.KeyAvailable) Console.ReadKey(true);

    while (true)
    {
        if (Console.KeyAvailable)
        {
            var k = Console.ReadKey(true);
            if (k.Key == ConsoleKey.Escape) return -1;
        }
        foreach (int vk in ALL_KEYS)
        {
            if ((GetAsyncKeyState(vk) & 0x8000) != 0)
            {
                while ((GetAsyncKeyState(vk) & 0x8000) != 0) Thread.Sleep(10);
                return vk;
            }
        }
        Thread.Sleep(10);
    }
}

double GetNormalRandom(double mean, double stdDev)
{
    double u1 = 1.0 - rng.NextDouble();
    double u2 = 1.0 - rng.NextDouble();
    double randStdNormal = Math.Sqrt(-2.0 * Math.Log(u1)) * Math.Sin(2.0 * Math.PI * u2);
    return mean + stdDev * randStdNormal;
}

string GetKeyName(int vk)
{
    switch (vk)
    {
        case 0x01: return "LButton";
        case 0x02: return "RButton";
        case 0x04: return "MButton";
        case 0x05: return "X1";
        case 0x06: return "X2";
        case 0x10: return "Shift";
        case 0x11: return "Ctrl";
        case 0x12: return "Alt";
        case 0x08: return "Back";
        case 0x09: return "Tab";
        case 0x0D: return "Enter";
        case 0x20: return "Space";
        case 0x21: return "PgUp";
        case 0x22: return "PgDn";
        case 0x23: return "End";
        case 0x24: return "Home";
        case 0x25: return "Left";
        case 0x26: return "Up";
        case 0x27: return "Right";
        case 0x28: return "Down";
        case 0x2C: return "PrtSc";
        case 0x2D: return "Ins";
        case 0x2E: return "Del";
        default:
            if (vk >= 0x30 && vk <= 0x39) return ((char)vk).ToString();
            if (vk >= 0x41 && vk <= 0x5A) return ((char)vk).ToString();
            if (vk >= 0x60 && vk <= 0x69) return "Num" + (vk - 0x60);
            if (vk >= 0x70 && vk <= 0x7B) return "F" + (vk - 0x70 + 1);
            return $"VK_0x{vk:X}";
    }
}

string T(string key)
{
    if (Translations.LANG.TryGetValue(key, out var dict))
    {
        if (dict.TryGetValue(config.language, out var trans)) return trans;
        if (dict.TryGetValue(Language.EN, out var def)) return def;
    }
    return $"???{key}???";
}

void LoadConfig()
{
    try
    {
        if (!File.Exists(CONFIG_FILE)) return;
        foreach (var line in File.ReadAllLines(CONFIG_FILE))
        {
            var parts = line.Split('=', 2);
            if (parts.Length < 2) continue;
            string key = parts[0].Trim();
            string value = parts[1].Trim();

            try
            {
                if (key == "BaseDelay") config.baseDelay = int.Parse(value);
                else if (key == "AntiDetect") config.antiDetect = value == "1";
                else if (key == "RandomRange") config.randomRange = int.Parse(value);
                else if (key == "RandomDist") config.randomDist = (RandomDist)int.Parse(value);
                else if (key == "HoldTime") config.holdTime = int.Parse(value);
                else if (key == "HoldRange") config.holdRange = int.Parse(value);
                else if (key == "ClickButton") config.clickButton = (ClickType)int.Parse(value);
                else if (key == "ClickPattern") config.clickPattern = (ClickPattern)int.Parse(value);
                else if (key == "BurstCount") config.burstCount = int.Parse(value);
                else if (key == "BurstInterval") config.burstInterval = int.Parse(value);
                else if (key == "BurstIntervalRange") config.burstIntervalRange = int.Parse(value);
                else if (key == "SimulateMove") config.simulateMove = value == "1";
                else if (key == "MoveRadius") config.moveRadius = int.Parse(value);
                else if (key == "TriggerKey") config.triggerKey = int.Parse(value);
                else if (key == "TriggerCount") config.triggerCount = int.Parse(value);
                else if (key == "TriggerTimeoutMs") config.triggerTimeoutMs = int.Parse(value);
                else if (key == "RandomBreakEnabled") config.randomBreakEnabled = value == "1";
                else if (key == "RandomBreakProbability") config.randomBreakProbability = int.Parse(value);
                else if (key == "RandomBreakMin") config.randomBreakMin = int.Parse(value);
                else if (key == "RandomBreakMax") config.randomBreakMax = int.Parse(value);
                else if (key == "Language") config.language = (Language)int.Parse(value);
            }
            catch { }
        }
    }
    catch { }
}
